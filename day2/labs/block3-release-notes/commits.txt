=== commit 5ab2e61 | 2026-10-15 | ravi
feat: webhook retries with exponential backoff
Failed webhook deliveries are retried up to 5 times (1s, 2s, 4s, 8s, 16s) before being marked failed.
files: api/webhooks.py, tests/test_webhooks.py
--- a/api/webhooks.py
+++ b/api/webhooks.py
@@ -22,8 +22,14 @@ def deliver(hook, payload):
-    resp = requests.post(hook.url, json=payload, timeout=5)
-    return resp.ok
+    for attempt in range(5):
+        resp = requests.post(hook.url, json=payload, timeout=5)
+        if resp.ok:
+            return True
+        time.sleep(2 ** attempt)
+    return False
